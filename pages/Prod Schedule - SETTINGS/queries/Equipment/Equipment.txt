SELECT
    e.id,
    e.make,
    e.model,
    e.description,
    e.active,
    COUNT(t.equipment_id) AS row_count,
    SUM(COALESCE(t.hours, 0)) AS total_hours
FROM mv_commercial.MV11_Workshop_equipment e
LEFT JOIN mv_commercial.MV11_Workshop_Work_Targets t
    ON t.equipment_id = e.id
WHERE e.active = {{!Equipment_Disabled_Switch.isSwitchedOn}}
GROUP BY
    e.id,
    e.make,
    e.model,
    e.description,
    e.active
ORDER BY e.description ASC;
